# FreeBSD 15.1-RELEASE 소스 코드 다운로드 (git이 설치되어 있어야 합니다)
sudo git clone -b releng/15.1 https://freebsd.org /usr/src

<LeftMouse># 1. Ports 트리 다운로드 및 최신화 (최초 1회)
sudo git clone https://freebsd.org /usr/ports

# 2. NVIDIA 드라이버 메인 포트로 이동하여 컴파일 및 설치
cd /usr/ports/x11/nvidia-driver
sudo make BATCH=yes install clean

# 3. DRM(Direct Rendering Manager) 커널 모듈 컴파일 및 설치
# (최신 그래픽 카드 및 Wayland/현대적인 데스크톱 환경 필수)
cd /usr/ports/graphics/nvidia-drm-kmod
sudo make install clean

# NVIDIA 드라이버 모듈을 부팅 시 자동 로드하도록 등록
sudo sysrc kld_list+="nvidia-modeset nvidia-drm"

# 리눅스 프로그램 호환성 및 관련 라이브러리 활성화
sudo sysrc linux_enable="YES"
